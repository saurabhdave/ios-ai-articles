### Swift Snippet of the Week

```swift
import Foundation
import SwiftUI
import Observation

@MainActor @Observable class FeedViewModel {
    var items: [String] = []
    private var currentTask: Task<Void, Never>?

    func loadParallelFeeds(urls: [URL]) {
        currentTask?.cancel()
        currentTask = Task { [weak self] in
            guard let self else { return }
            await withTaskGroup(of: String?.self) { group in
                for url in urls { group.addTask { try? await self.fetch(url: url) } }
                var results: [String] = []
                for await piece in group {
                    if Task.isCancelled { break }
                    if let p = piece { results.append(p) }
                }
                self.items = results
            }
        }
    }

    func cancelCurrent() { currentTask?.cancel(); currentTask = nil }
// … (truncated for newsletter)
```

This pattern encodes the pragmatic judgment: scope async work to the UI owner, cancel previous work on navigation, and use TaskGroup for cooperative parallelism so you avoid orphaned background activity.
